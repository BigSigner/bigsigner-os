start_file=start_x.elf
fixup_file=fixup_x.dat

# Kernel location
kernel=zImage

# How much memory in MB to assign to the GPU on Pi models having
gpu_mem=64

# Enable Camera module
start_x=1

# Enable SPI
dtparam=spi=on

# Set the bootloader delay to 0 seconds. The default is 1s if not specified
boot_delay=0
disable_splash=1

### Dev image differences from the release boot_config.txt ###

# Wifi and bluetooth stay enabled (release adds disable-wifi/disable-bt here)

# USB relay: expose the USB port as an ethernet gadget when plugged into a
# computer (see docs/dev_images.md). dwc2 is built dual-role, so a keyboard
# on an OTG adapter still works.
dtoverlay=dwc2

# HDMI console output even when the display is plugged in after boot.
# Waveshare 4inch HDMI LCD (SKU 12030): 480x800 portrait.
# Touch is the XPT2046 on SPI CS1, interrupt GPIO25.
hdmi_force_hotplug=1
hdmi_group=2
hdmi_mode=87
hdmi_cvt=480 800 60 6 0 0 0
hdmi_drive=1
disable_overscan=1
framebuffer_width=480
framebuffer_height=800
# 16-bit RGB565. A 32-bit framebuffer makes the app rebuild every pixel in
# software before HDMI can show it. 16-bit halves that copy.
framebuffer_depth=16
dtoverlay=ads7846,cs=1,penirq=25,penirq_pull=2,speed=50000,keep_vref_on=0,swapxy=0,pmax=255,xohms=150,xmin=400,xmax=3800,ymin=400,ymax=3800
